// 데이터 최초 로드 (redirect: "follow" 옵션 추가로 CORS/302 문제 방지) async function fetchQuestions() { const startBtn = document.getElementById('start-btn'); const examSelect = document.getElementById('exam-select'); try { const res = await fetch(API_URL, { method: "GET", redirect: "follow" }); const json = await res.json(); if(json.status === "success" && json.data && json.data.length > 0) { allQuestions = json.data; // 회차 이름 추출 (공백 제거 및 중복 제거) const examTitles = [...new Set(allQuestions.map(q => String(q.exam_title || '').trim()))].filter(Boolean); examSelect.innerHTML = ''; examTitles.forEach(title => { const opt = document.createElement('option'); opt.value = title; opt.innerText = title; examSelect.appendChild(opt); }); startBtn.disabled = false; startBtn.innerText = "시험 시작하기"; } else { examSelect.innerHTML = ''; startBtn.innerText = "데이터가 없습니다"; } } catch(e) { console.error(e); examSelect.innerHTML = ''; startBtn.innerText = "로딩 실패 (새로고침 하세요)"; } }
